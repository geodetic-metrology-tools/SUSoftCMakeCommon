# Plateform

##########################################
# Operating system
# Set the path to the external libraries if it has not been set already 
if(NOT EXT_LIB_PATH)
	if(UNIX)
		set(EXT_LIB_PATH					"/susoft/ext")
	else()
		set(EXT_LIB_PATH					"C:/susoft/ext")
	endif()
endif()

if(UNIX)
	set(QT_ROOT_PATH					"/usr/share/qt5")
endif()

##########################################
# Compiled version
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
	MESSAGE("-- 64 bits platform detected --")
	set(PLATFORM 64)
else(CMAKE_SIZEOF_VOID_P EQUAL 4)
	MESSAGE("-- 32 bits platform detected --")
	set(PLATFORM 32)
endif()

##########################################
#
# Librairies
#
set(BOOST_ROOT							"${EXT_LIB_PATH}/boost")
set(EIGEN_INCLUDE_PATH					"${EXT_LIB_PATH}/eigen")
set(TCLAP_INCLUDE_PATH					"${EXT_LIB_PATH}/tclap/include")
set(TUT_INCLUDE_PATH					"${EXT_LIB_PATH}/tut-framework/include")

##########################################
#
# WINDOWS
#
if(WIN32)

	##########################################
	# Qt
	# Search for the first fitting available Qt 
	file(GLOB QT_VERSION_DIR RELATIVE C:/Qt C:/Qt/5.15*)
	if(PLATFORM EQUAL 64)
		set(QT_ROOT_PATH				"C:/Qt/${QT_VERSION_DIR}/msvc2019_64")
	else()
		set(QT_ROOT_PATH				"C:/Qt/${QT_VERSION_DIR}/msvc2019")
	endif()
	set(QT_VERSION_MAJOR				5)

	##########################################
	# OpenSSL
	if(PLATFORM EQUAL 64)
		set(OPENSSL_ROOT_DIR				"${QT_ROOT_PATH}/../../Tools/OpenSSL/Win_x64")
	else()
		set(OPENSSL_ROOT_DIR				"${QT_ROOT_PATH}/../../Tools/OpenSSL/Win_x32")
	endif()

	##########################################
	# Other susoft related
	if(PLATFORM EQUAL 64)
		set(REFRAME2016_LIBRARY_DIR			"${EXT_LIB_PATH}/Reframe2016/x64")
	else()
		set(REFRAME2016_LIBRARY_DIR			"${EXT_LIB_PATH}/Reframe2016/x86")
	endif()
	
endif()
